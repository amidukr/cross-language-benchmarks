cmake_minimum_required(VERSION 3.18)
project(enum_dispatch LANGUAGES C CXX)
add_executable(enum-dispatch-c c/main.c)
add_executable(enum-dispatch-cpp cpp/main.cpp)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(enum-dispatch-c PRIVATE -std=c17 -O3 -DNDEBUG -Wall -Wextra -Wpedantic)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(enum-dispatch-cpp PRIVATE -std=c++20 -O3 -DNDEBUG -Wall -Wextra -Wpedantic)
endif()
